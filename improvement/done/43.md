1. **Create LoggerService** (`src/services/LoggerService.js`):
   ```javascript
   import { invoke } from '@tauri-apps/api/core';

   class LoggerService {
     constructor() {
       this.logs = [];
       this.maxLogs = 1000;
       this.sessionId = `sess_${Date.now()}_${Math.random().toString(36).substr(2, 9)}`;
       this.correlationCounter = 0;
     }

     generateCorrelationId() {
       return `frontend_corr_${this.sessionId}_${++this.correlationCounter}`;
     }

     log(level, component, event, message, data = {}) {
       const logEntry = {
         timestamp: new Date().toISOString(),
         sessionId: this.sessionId,
         level,
         component,
         event,
         message,
         data,
         correlationId: data.correlationId || this.generateCorrelationId()
       };

       this.logs.push(logEntry);
       if (this.logs.length > this.maxLogs) {
         this.logs.shift();
       }

       // Console fallback for development
       if (process.env.NODE_ENV === 'development') {
         console.log(`[${level}] ${component}:${event} - ${message}`, data);
       }
     }

     debug(component, event, message, data = {}) {
       this.log('DEBUG', component, event, message, data);
     }

     info(component, event, message, data = {}) {
       this.log('INFO', component, event, message, data);
     }

     warn(component, event, message, data = {}) {
       this.log('WARN', component, event, message, data);
     }

     error(component, event, message, data = {}) {
       this.log('ERROR', component, event, message, data);
     }

     async flushToBackend() {
       if (this.logs.length === 0) return;
       
       try {
         await invoke('submit_frontend_logs', {
           logs: JSON.stringify(this.logs)
         });
         this.logs = [];
       } catch (error) {
         console.error('Failed to flush logs to backend:', error);
       }
     }

     getLogs(filter = {}) {
       return this.logs.filter(log => {
         if (filter.level && log.level !== filter.level) return false;
         if (filter.component && log.component !== filter.component) return false;
         if (filter.since && new Date(log.timestamp) < new Date(filter.since)) return false;
         return true;
       });
     }
   }

   export const logger = new LoggerService();
   export default LoggerService;
   ```

1. **Create LogViewer component** (`src/App/LogViewer.jsx`):
   ```javascript
   import React, { useState, useEffect } from 'react';
   import { invoke } from '@tauri-apps/api/core';
   import { logger } from '../services/LoggerService.js';
   import LogTable from '../components/LogTable.jsx';
   import LogFilters from '../components/LogFilters.jsx';

   const LogViewer = () => {
     const [logs, setLogs] = useState([]);
     const [backendLogs, setBackendLogs] = useState('');
     const [filters, setFilters] = useState({
       level: 'all',
       component: 'all',
       since: '1h',
       source: 'all' // 'frontend', 'backend', 'all'
     });

     const loadLogs = async () => {
       try {
         // Get frontend logs
         const frontendLogs = logger.getLogs(filters);
         
         // Get backend logs
         const backendLogsResult = await invoke('get_logs', {
           logType: 'all',
           lines: 1000,
           since: filters.since
         });

         setLogs(frontendLogs);
         setBackendLogs(backendLogsResult);
       } catch (error) {
         console.error('Failed to load logs:', error);
       }
     };

     useEffect(() => {
       loadLogs();
       const interval = setInterval(loadLogs, 5000); // Refresh every 5 seconds
       return () => clearInterval(interval);
     }, [filters]);

     const exportLogs = () => {
       const allLogs = {
         frontend: logs,
         backend: backendLogs,
         timestamp: new Date().toISOString(),
         filters
       };

       const blob = new Blob([JSON.stringify(allLogs, null, 2)], {
         type: 'application/json'
       });
       
       const url = URL.createObjectURL(blob);
       const link = document.createElement('a');
       link.href = url;
       link.download = `photoclove-logs-${new Date().toISOString().split('T')[0]}.json`;
       link.click();
       URL.revokeObjectURL(url);
     };

     return (
       <div className="log-viewer">
         <div className="log-viewer-header">
           <h2>Debug Logs</h2>
           <button onClick={exportLogs}>Export Logs</button>
           <button onClick={loadLogs}>Refresh</button>
         </div>
         
         <LogFilters 
           filters={filters} 
           onFilterChange={setFilters} 
         />
         
         <div className="log-content">
           <LogTable 
             logs={logs} 
             backendLogs={backendLogs}
             filters={filters}
           />
         </div>
       </div>
     );
   };

   export default LogViewer;
   ```
